package booksearch.vo;

//VO
public class Book {

    private String bisbn;
    private String btitle;
    private int bprice;
    private String bauthor;

    public Book() {

    }

    public Book(String bisbn, String btitle, int bprice, String bauthor) {
        this.bisbn = bisbn;
        this.btitle = btitle;
        this.bprice = bprice;
        this.bauthor = bauthor;
    }

    public String getBisbn() {
        return bisbn;
    }

    public void setBisbn(String bisbn) {
        this.bisbn = bisbn;
    }

    public String getBtitle() {
        return btitle;
    }

    public void setBtitle(String btitle) {
        this.btitle = btitle;
    }

    public int getBprice() {
        return bprice;
    }

    public void setBprice(int bprice) {
        this.bprice = bprice;
    }

    public String getBauthor() {
        return bauthor;
    }

    public void setBauthor(String bauthor) {
        this.bauthor = bauthor;
    }
}

package booksearch;

import javafx.collections.FXCollections;
import javafx.collections.ObservableList;
import javafx.fxml.FXML;
import javafx.fxml.Initializable;
import javafx.scene.control.*;
import javafx.scene.control.cell.PropertyValueFactory;
import javafx.scene.layout.GridPane;
import booksearch.vo.Book;

import java.net.URL;
import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;
import java.util.Optional;
import java.util.ResourceBundle;

public class JavaFXController implements Initializable {

    @FXML
    private TableView<Book> tableView;

    @FXML private Button searchBtn;
    @FXML private Button editBtn;
    @FXML private TableColumn<Book, String> isbnCol;
    @FXML private TableColumn<Book, String> titleCol;
    @FXML private TableColumn<Book, Integer> priceCol;
    @FXML private TableColumn<Book, String> authorCol;
    @FXML private TextField searchField;
    @FXML private Button insertBtn;

    @Override
    public void initialize(URL url, ResourceBundle resourceBundle) {
        // 테이블 컬럼과 Book 객체의 속성을 매핑
        isbnCol.setCellValueFactory(new PropertyValueFactory<>("bisbn"));
        titleCol.setCellValueFactory(new PropertyValueFactory<>("btitle"));
        priceCol.setCellValueFactory(new PropertyValueFactory<>("bprice"));
        authorCol.setCellValueFactory(new PropertyValueFactory<>("bauthor"));

        // 검색 버튼 클릭 시 책 목록 로드
        searchBtn.setOnAction(event -> loadBooksFromDB(searchField.getText()));

        // 삽입 버튼 클릭 시 책 추가 다이얼로그 열기
        insertBtn.setOnAction(event -> handleInsertButtonAction());

        // 테이블에서 더블클릭 시 수정 기능 호출
        tableView.setOnMouseClicked(event -> {
            if (event.getClickCount() == 2) {  // 더블클릭
                handleEditButtonAction();
            }
        });

        // 우클릭 시 삭제 메뉴가 뜨도록 ContextMenu 설정
        ContextMenu contextMenu = new ContextMenu();
        MenuItem deleteMenuItem = new MenuItem("삭제");
        deleteMenuItem.setOnAction(event -> handleDeleteButtonAction());
        contextMenu.getItems().add(deleteMenuItem);

        tableView.setContextMenu(contextMenu);

        // 초기화 시 자동으로 책 목록을 로드
        loadBooksFromDB(searchField.getText());
    }

    // 데이터베이스에서 책 목록 로드
    private void loadBooksFromDB(String keyword) {
        ObservableList<Book> books = FXCollections.observableArrayList();
        String url = "jdbc:oracle:thin:@localhost:1521:xe";
        String id = "C##DEV";
        String pw = "sys1234";
        String sql = "SELECT bisbn, btitle, bprice, bauthor FROM book WHERE btitle LIKE ?";

        try (Connection con = DriverManager.getConnection(url, id, pw);
             PreparedStatement pstmt = con.prepareStatement(sql)) {

            pstmt.setString(1, "%" + keyword + "%");  // 검색 키워드 설정
            ResultSet rs = pstmt.executeQuery();

            while (rs.next()) {
                books.add(new Book(
                        rs.getString("bisbn"),
                        rs.getString("btitle"),
                        rs.getInt("bprice"),
                        rs.getString("bauthor")
                ));
            }
            rs.close();
            pstmt.close();
            con.close();
        } catch (Exception e) {
            e.printStackTrace();
        }

        // 테이블에 책 목록 세팅
        tableView.setItems(books);
    }

    // 책을 데이터베이스에 추가
    private void insertBookIntoDB(Book book) {
        String url = "jdbc:oracle:thin:@localhost:1521:xe";
        String id = "C##DEV";
        String pw = "sys1234";
        String sql = "INSERT INTO book (bisbn, btitle, bprice, bauthor) VALUES (?, ?, ?, ?)";

        try (Connection con = DriverManager.getConnection(url, id, pw);
             PreparedStatement pstmt = con.prepareStatement(sql)) {

            pstmt.setString(1, book.getBisbn());
            pstmt.setString(2, book.getBtitle());
            pstmt.setInt(3, book.getBprice());
            pstmt.setString(4, book.getBauthor());
            pstmt.executeUpdate();
            pstmt.close();
            con.close();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    // 삭제 버튼 클릭 시 선택한 책 삭제
    @FXML
    private void handleDeleteButtonAction() {
        Book selectedBook = tableView.getSelectionModel().getSelectedItem();
        if (selectedBook != null) {
            deleteBookFromDB(selectedBook.getBisbn());
            loadBooksFromDB(searchField.getText());  // 삭제 후 테이블 갱신
        }
    }

    // 책 추가 다이얼로그 처리
    @FXML
    private void handleInsertButtonAction() {
        // 새 책 정보를 입력하는 다이얼로그 생성
        Dialog<Book> dialog = new Dialog<>();
        dialog.setTitle("새 책 추가");
        dialog.setHeaderText("새 책 정보를 입력하세요.");
        dialog.getDialogPane().getButtonTypes().addAll(ButtonType.OK, ButtonType.CANCEL);

        // 다이얼로그에 입력 필드 추가
        GridPane grid = new GridPane();
        grid.setHgap(10);
        grid.setVgap(10);

        TextField isbnField = new TextField();
        TextField titleField = new TextField();
        TextField priceField = new TextField();
        TextField authorField = new TextField();

        grid.add(new Label("ISBN:"), 0, 0);
        grid.add(isbnField, 1, 0);
        grid.add(new Label("책 제목:"), 0, 1);
        grid.add(titleField, 1, 1);
        grid.add(new Label("책 가격:"), 0, 2);
        grid.add(priceField, 1, 2);
        grid.add(new Label("책 저자:"), 0, 3);
        grid.add(authorField, 1, 3);

        dialog.getDialogPane().setContent(grid);

        // 다이얼로그 결과 처리
        dialog.setResultConverter(buttonType -> {
            if (buttonType == ButtonType.OK) {
                return new Book(
                        isbnField.getText(),
                        titleField.getText(),
                        Integer.parseInt(priceField.getText()),
                        authorField.getText()
                );
            }
            return null;
        });

        Optional<Book> result = dialog.showAndWait();
        result.ifPresent(book -> {
            insertBookIntoDB(book);
            loadBooksFromDB(searchField.getText());  // 새로 추가된 책 반영
        });
    }

    // 데이터베이스에서 책 삭제
    private void deleteBookFromDB(String bisbn) {
        String url = "jdbc:oracle:thin:@localhost:1521:xe";
        String id = "C##DEV";
        String pw = "sys1234";
        String sql = "DELETE FROM book WHERE bisbn = ?";

        try (Connection con = DriverManager.getConnection(url, id, pw);
             PreparedStatement pstmt = con.prepareStatement(sql)) {

            pstmt.setString(1, bisbn);
            pstmt.executeUpdate();
            pstmt.close();
            con.close();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }

    // 수정 버튼 클릭 시 책 수정 다이얼로그 열기
    @FXML
    private void handleEditButtonAction() {
        Book selectedBook = tableView.getSelectionModel().getSelectedItem();
        if (selectedBook != null) {
            // 책 정보를 수정하는 다이얼로그 생성
            Dialog<Book> dialog = new Dialog<>();
            dialog.setTitle("책 정보 수정");
            dialog.setHeaderText("책 정보를 수정하세요.");

            // 다이얼로그 버튼 설정
            dialog.getDialogPane().getButtonTypes().addAll(ButtonType.OK, ButtonType.CANCEL);

            // 다이얼로그 내용 설정
            GridPane grid = new GridPane();
            grid.setHgap(10);
            grid.setVgap(10);

            TextField isbnField = new TextField(selectedBook.getBisbn());
            TextField titleField = new TextField(selectedBook.getBtitle());
            TextField priceField = new TextField(String.valueOf(selectedBook.getBprice()));
            TextField authorField = new TextField(selectedBook.getBauthor());

            grid.add(new Label("ISBN:"), 0, 0);
            grid.add(isbnField, 1, 0);
            grid.add(new Label("책 제목:"), 0, 1);
            grid.add(titleField, 1, 1);
            grid.add(new Label("책 가격:"), 0, 2);
            grid.add(priceField, 1, 2);
            grid.add(new Label("책 저자:"), 0, 3);
            grid.add(authorField, 1, 3);

            dialog.getDialogPane().setContent(grid);

            // 다이얼로그 결과 처리
            dialog.setResultConverter(buttonType -> {
                if (buttonType == ButtonType.OK) {
                    return new Book(
                            isbnField.getText(),
                            titleField.getText(),
                            Integer.parseInt(priceField.getText()),
                            authorField.getText()
                    );
                }
                return null;
            });

            Optional<Book> result = dialog.showAndWait();
            result.ifPresent(updatedBook -> {
                updateBookInDB(updatedBook);
                loadBooksFromDB(searchField.getText());  // 수정 후 테이블 갱신
            });
        }
    }

    // 데이터베이스에서 책 정보 수정
    private void updateBookInDB(Book book) {
        String url = "jdbc:oracle:thin:@localhost:1521:xe";
        String id = "C##DEV";
        String pw = "sys1234";
        String sql = "UPDATE book SET btitle = ?, bprice = ?, bauthor = ? WHERE bisbn = ?";

        try (Connection con = DriverManager.getConnection(url, id, pw);
             PreparedStatement pstmt = con.prepareStatement(sql)) {

            pstmt.setString(1, book.getBtitle());
            pstmt.setInt(2, book.getBprice());
            pstmt.setString(3, book.getBauthor());
            pstmt.setString(4, book.getBisbn());
            pstmt.executeUpdate();
            pstmt.close();
            con.close();
        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}

package booksearch;

import javafx.application.Application;
import javafx.fxml.FXMLLoader;
import javafx.scene.Parent;
import javafx.scene.Scene;
import javafx.stage.Stage;

public class JavaFXTableView extends Application {

    @Override
    public void start(Stage stage) throws Exception {
        //화면 구성
        Parent root = null;
        FXMLLoader loader = new FXMLLoader(getClass().getResource("/booksearch/table.fxml"));

        try {
            root = loader.load();
        } catch (Exception e) {
            e.printStackTrace();
        }

        Scene scene = new Scene(root);
        stage.setScene(scene);
        stage.show();
    }

    public static void main(String[] args) {
        launch(args);
    }
}
