SOR시스템 개발중
```
#define _CRT_SECURE_NO_WARNINGS
#include "Env.h"
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define KRX_CHARGE 1.5
#define ATS_CHARGE 1.45

typedef struct {
    int id;
    int offernumber;
    int exchangeactual;
    char selleraccountnumber[20];
    char buyeraccountnumber[20];
    int quantity;
    int price;
    int charge;
    char createdAt[20];
} TRADE;

TRADE* sor(char stockcode[], char status[], int price, int quantity, int* trade_count) {
    printf("sor실행\n");
    *trade_count = 0;
    int max_trades = 10; // 초기 거래 저장 공간
    TRADE* trades = (TRADE*)malloc(max_trades * sizeof(TRADE));

    if (trades == NULL) {
        printf("메모리 할당 실패\n");
        return NULL;
    }

    set_env(); // 환경 설정


    char select_sql[512];
    sprintf(select_sql,
        "SELECT STOCK_CODE, TRADING_TIME, "
        "SELL_PRICE_1, SELL_QUANTITY_1, BUY_PRICE_1, BUY_QUANTITY_1, "
        "ATS_SELL_QUANTITY_1, ATS_BUY_QUANTITY_1 "
        "FROM STOCKASKINGPRICE WHERE STOCK_CODE = '%s'", stockcode);

    OCIHandleAlloc(envhp, (void**)&stmthp, OCI_HTYPE_STMT, 0, NULL);
    OCIStmtPrepare(stmthp, errhp, (text*)select_sql, strlen(select_sql), OCI_NTV_SYNTAX, OCI_DEFAULT);
    OCIStmtExecute(svchp, stmthp, errhp, 1, 0, NULL, NULL, OCI_DEFAULT); // 인수 개수 수정

    int sell_price, buy_price;
    int sell_quantity, buy_quantity;
    int ats_sell_quantity, ats_buy_quantity;

    OCIDefine* def_sell_price, * def_sell_quantity;
    OCIDefine* def_buy_price, * def_buy_quantity;
    OCIDefine* def_ats_sell_quantity, * def_ats_buy_quantity;

    OCIDefineByPos(stmthp, &def_sell_price, errhp, 3, &sell_price, sizeof(int), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def_sell_quantity, errhp, 4, &sell_quantity, sizeof(int), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def_buy_price, errhp, 5, &buy_price, sizeof(int), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def_buy_quantity, errhp, 6, &buy_quantity, sizeof(int), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def_ats_sell_quantity, errhp, 7, &ats_sell_quantity, sizeof(int), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);
    OCIDefineByPos(stmthp, &def_ats_buy_quantity, errhp, 8, &ats_buy_quantity, sizeof(int), SQLT_INT, NULL, NULL, NULL, OCI_DEFAULT);

    while (OCIStmtFetch2(stmthp, errhp, 1, OCI_DEFAULT, 0, OCI_DEFAULT) == OCI_SUCCESS) {
        int remaining_qty = quantity;

        if (strcmp(status, "매수") == 0) {
            // ATS 매도 주문 먼저 처리
            if (remaining_qty > 0 && ats_sell_quantity > 0) {
                int trade_qty = (remaining_qty < ats_sell_quantity) ? remaining_qty : ats_sell_quantity;

                if (*trade_count >= max_trades) {
                    max_trades *= 2;
                    trades = (TRADE*)realloc(trades, max_trades * sizeof(TRADE));
                }

                trades[*trade_count] = (TRADE){
                    .quantity = trade_qty,
                    .price = sell_price,
                    .exchangeactual = 1,
                    .charge = trade_qty * ATS_CHARGE
                };

                // 디버깅을 위한 printf 추가
                printf("Trade #%d: Quantity=%d, Price=%d, ExchangeActual=%d, Charge=%.2f\n",
                    *trade_count, trades[*trade_count].quantity, trades[*trade_count].price,
                    trades[*trade_count].exchangeactual, trades[*trade_count].charge);

                (*trade_count)++;
                remaining_qty -= trade_qty;
            }

            // 정규거래소 매도 주문 처리
            if (remaining_qty > 0 && sell_quantity > 0) {
                int trade_qty = (remaining_qty < sell_quantity) ? remaining_qty : sell_quantity;

                if (*trade_count >= max_trades) {
                    max_trades *= 2;
                    trades = (TRADE*)realloc(trades, max_trades * sizeof(TRADE));
                }

                trades[*trade_count] = (TRADE){
                    .quantity = trade_qty,
                    .price = sell_price,
                    .exchangeactual = 0,
                    .charge = trade_qty * KRX_CHARGE
                };

                // 디버깅을 위한 printf 추가
                printf("Trade #%d: Quantity=%d, Price=%d, ExchangeActual=%d, Charge=%.2f\n",
                    *trade_count, trades[*trade_count].quantity, trades[*trade_count].price,
                    trades[*trade_count].exchangeactual, trades[*trade_count].charge);

                (*trade_count)++;
                remaining_qty -= trade_qty;
            }
        }
        else if (strcmp(status, "매도") == 0) {
            // ATS 매수 주문 처리
            if (remaining_qty > 0 && ats_buy_quantity > 0) {
                int trade_qty = (remaining_qty < ats_buy_quantity) ? remaining_qty : ats_buy_quantity;

                if (*trade_count >= max_trades) {
                    max_trades *= 2;
                    trades = (TRADE*)realloc(trades, max_trades * sizeof(TRADE));
                }

                trades[*trade_count] = (TRADE){
                    .quantity = trade_qty,
                    .price = buy_price,
                    .exchangeactual = 1,
                    .charge = trade_qty * ATS_CHARGE
                };

                // 디버깅을 위한 printf 추가
                printf("Trade #%d: Quantity=%d, Price=%d, ExchangeActual=%d, Charge=%.2f\n",
                    *trade_count, trades[*trade_count].quantity, trades[*trade_count].price,
                    trades[*trade_count].exchangeactual, trades[*trade_count].charge);

                (*trade_count)++;
                remaining_qty -= trade_qty;
            }

            // 정규거래소 매수 주문 처리
            if (remaining_qty > 0 && buy_quantity > 0) {
                int trade_qty = (remaining_qty < buy_quantity) ? remaining_qty : buy_quantity;

                if (*trade_count >= max_trades) {
                    max_trades *= 2;
                    trades = (TRADE*)realloc(trades, max_trades * sizeof(TRADE));
                }

                trades[*trade_count] = (TRADE){
                    .quantity = trade_qty,
                    .price = buy_price,
                    .exchangeactual = 0,
                    .charge = trade_qty * KRX_CHARGE
                };

                // 디버깅을 위한 printf 추가
                printf("Trade #%d: Quantity=%d, Price=%d, ExchangeActual=%d, Charge=%.2f\n",
                    *trade_count, trades[*trade_count].quantity, trades[*trade_count].price,
                    trades[*trade_count].exchangeactual, trades[*trade_count].charge);

                (*trade_count)++;
                remaining_qty -= trade_qty;
            }
        }

        if (remaining_qty == 0) break;
    }

    quit_env(); // 환경 종료
    return trades;
}
```
