// 배열 원소의 합을 구하는 함수
void sumArray(int **arr) {
    int sum = 0;

void rotate90(int (*arr)[SIZE]) {
    int rotated[SIZE][SIZE];

    for (int i = 0; i < SIZE; i++) {
        for (int j = 0; j < SIZE; j++) {
            rotated[j][SIZE - 1 - i] = arr[i][j];
        }
    }

    printf("90도 회전된 행렬: ");
    for (int i = 0; i < SIZE; i++) {
        for (int j = 0; j < SIZE; j++) {
            printf("%d ", rotated[i][j]);
        }
        printf("\n");
    }
}

}

void sort(int arr[], int size, int (*compare)(int, int)) {
    for (int i = 0; i < size - 1; i++) {
        for (int j = 0; j < size - 1 - i; j++) {
            if (compare(arr[j], arr[j + 1])) {
                int temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }
}

#define MAX_SIZE 5

void fillArray(int arr[MAX_SIZE][MAX_SIZE], int rows, int cols) {

}

void printArray(int arr[MAX_SIZE][MAX_SIZE], int rows, int cols) {

    }

void deleteRow(int arr[MAX_SIZE][MAX_SIZE], int *rows, int cols, int delRow) {
    if (delRow < 0 || delRow >= *rows) {
        printf("잘못된 행 번호입니다.\n");
        return;
    }

    for (int i = delRow; i < (*rows) - 1; i++) {
        for (int j = 0; j < cols; j++) {
            arr[i][j] = arr[i + 1][j];
        }
    }
    (*rows)--;
}
