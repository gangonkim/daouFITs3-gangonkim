#include <stdio.h>
#include <stdlib.h>

#define MAX_NAME_LEN 20

// 대출 정보를 저장할 구조체
typedef struct {
    double loan_amount;
    double interest_rate;
    int loan_period;
    double remaining_balance;
} Loan;

// 고객 정보를 저장할 구조체
typedef struct {
    char name[MAX_NAME_LEN];
    int account_number;
    int loan_count;
    Loan* loans;
} Customer;

int main(void) {
    int num;
    printf("고객 수를 입력하세요: ");
    scanf_s("%d", &num);

    Customer* customers = malloc(sizeof(Customer) * num);
    if (customers == NULL) {
        printf("메모리 할당 실패!\n");
        return 1;
    }

    for (int i = 0; i < num; i++) {
        printf("고객 %d의 이름을 입력하세요: ", i + 1);
        scanf_s("%s", customers[i].name, (unsigned)_countof(customers[i].name));

        printf("고객 %d의 계좌번호를 입력하세요: ", i + 1);
        scanf_s("%d", &customers[i].account_number);

        printf("고객 %d이 받은 대출 수를 입력하세요: ", i + 1);
        scanf_s("%d", &customers[i].loan_count);

        customers[i].loans = malloc(sizeof(Loan) * customers[i].loan_count);
        if (customers[i].loans == NULL) {
            printf("메모리 할당 실패!\n");
            return 1;
        }

        for (int j = 0; j < customers[i].loan_count; j++) {
            printf("대출 %d의 금액을 입력하세요: ", j + 1);
            scanf_s("%lf", &customers[i].loans[j].loan_amount);

            printf("대출 %d의 이자율을 입력하세요: ", j + 1);
            scanf_s("%lf", &customers[i].loans[j].interest_rate);

            printf("대출 %d의 기간(년)을 입력하세요: ", j + 1);
            scanf_s("%d", &customers[i].loans[j].loan_period);

            // 대출 금액을 초기화하여 remaining_balance 설정
            customers[i].loans[j].remaining_balance = customers[i].loans[j].loan_amount;
        }

        for (int j = 0; j < customers[i].loan_count; j++) {
            double amount;
            printf("대출 %d의 상환할 금액을 입력하세요: ", j + 1);
            scanf_s("%lf", &amount);
            customers[i].loans[j].remaining_balance -= amount;
        }
    }

    // 입력된 고객 정보 출력
    printf("\n입력된 고객 및 대출 정보:\n");
    for (int i = 0; i < num; i++) {
        printf("\n고객 %d: %s, 계좌번호: %d\n", i + 1, customers[i].name, customers[i].account_number);
        for (int j = 0; j < customers[i].loan_count; j++) {
            printf("  대출 %d - 금액: %.2lf, 이자율: %.2lf%%, 기간: %d년, 남은 잔고: %.2lf\n",
                   j + 1,
                   customers[i].loans[j].loan_amount,
                   customers[i].loans[j].interest_rate,
                   customers[i].loans[j].loan_period,
                   customers[i].loans[j].remaining_balance);
        }
    }

    // 동적 메모리 해제
    for (int i = 0; i < num; i++) {
        free(customers[i].loans);
    }
    free(customers);

    return 0;
}

#include <stdio.h>
#include <stdlib.h>

#define MAX_NAME_LEN 50
#define MAX_CUSTOMERS 100
#define MAX_TRADES 50

// 주식 거래 정보를 저장할 구조체
typedef struct {
    char stock_name[MAX_NAME_LEN];  // 주식 이름
    int quantity;                   // 거래 수량
    int price;                       // 거래 가격
} Trade;

// 고객 정보를 저장할 구조체
typedef struct {
    char name[MAX_NAME_LEN]; // 고객 이름
    int account_number;      // 계좌 번호
    int trade_count;         // 거래 횟수
    Trade* trades;           // 거래 내역 (동적 할당)
} Customer;

int main() {
    int num_customers;

    // 고객 수 입력
    do {
        printf("고객 수를 입력하세요 (1~100): ");
        scanf("%d", &num_customers);
        if (num_customers < 1 || num_customers > MAX_CUSTOMERS) {
            printf("잘못된 입력입니다. 1~100 사이의 값을 입력하세요.\n");
        }
    } while (num_customers < 1 || num_customers > MAX_CUSTOMERS);

    // 고객 배열 동적 할당
    Customer* customers = (Customer*)malloc(sizeof(Customer) * num_customers);
    if (customers == NULL) {
        printf("메모리 할당 실패\n");
        return 1;
    }

    // 고객 정보 입력
    for (int i = 0; i < num_customers; i++) {
        printf("\n고객 %d 정보 입력\n", i + 1);
        printf("이름: ");
        scanf("%s", customers[i].name);
        printf("계좌 번호: ");
        scanf("%d", &customers[i].account_number);

        // 거래 내역 입력
        do {
            printf("거래 횟수를 입력하세요 (1~50): ");
            scanf("%d", &customers[i].trade_count);
            if (customers[i].trade_count < 1 || customers[i].trade_count > MAX_TRADES) {
                printf("잘못된 입력입니다. 1~50 사이의 값을 입력하세요.\n");
            }
        } while (customers[i].trade_count < 1 || customers[i].trade_count > MAX_TRADES);

        // 거래 내역 동적 할당
        customers[i].trades = (Trade*)malloc(sizeof(Trade) * customers[i].trade_count);
        if (customers[i].trades == NULL) {
            printf("메모리 할당 실패\n");
            return 1;
        }

        // 각 거래 입력
        for (int j = 0; j < customers[i].trade_count; j++) {
            printf("\n거래 %d 정보 입력\n", j + 1);
            printf("주식 이름: ");
            scanf("%s", customers[i].trades[j].stock_name);
            printf("거래 수량: ");
            scanf("%d", &customers[i].trades[j].quantity);
            printf("거래 가격: ");
            scanf("%d", &customers[i].trades[j].price);
        }
    }

    // 입력된 정보 출력
    printf("\n=== 고객 및 주식 거래 내역 ===\n");
    for (int i = 0; i < num_customers; i++) {
        printf("\n고객 %d\n", i + 1);
        printf("이름: %s\n", customers[i].name);
        printf("계좌 번호: %d\n", customers[i].account_number);
        printf("거래 내역:\n");
        for (int j = 0; j < customers[i].trade_count; j++) {
            printf("  [%d] 주식: %s, 수량: %d, 가격: %d\n",
                   j + 1, customers[i].trades[j].stock_name,
                   customers[i].trades[j].quantity, customers[i].trades[j].price);
        }
    }

    // 동적 할당된 메모리 해제
    for (int i = 0; i < num_customers; i++) {
        free(customers[i].trades); // 거래 내역 해제
    }
    free(customers); // 고객 배열 해제

    return 0;
}

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_NAME_LEN 50
#define MAX_CUSTOMERS 100
#define MAX_STOCKS 50

// 주식 정보를 저장하는 구조체
typedef struct {
    char stock_name[MAX_NAME_LEN];  // 주식 이름
    double purchase_price;          // 평균 구매 가격
    double current_price;           // 현재 가격
    int quantity;                   // 보유 수량
} Stock;

// 고객 정보를 저장하는 구조체
typedef struct {
    char name[MAX_NAME_LEN]; // 고객 이름
    int account_number;      // 계좌 번호
    int stock_count;         // 보유 주식 개수
    Stock* portfolio;        // 주식 포트폴리오 (동적 할당)
} Customer;

void add_customer(Customer* customers, int* num_customers);
void add_portfolio(Customer* customer);
void display_customers(Customer* customers, int num_customers);
void free_memory(Customer* customers, int num_customers);

int main() {
    Customer customers[MAX_CUSTOMERS];
    int num_customers = 0;
    int choice;

    while (1) {
        printf("\n=== 증권사 고객 포트폴리오 관리 시스템 ===\n");
        printf("1. 고객 추가\n");
        printf("2. 고객 포트폴리오 추가 및 수정\n");
        printf("3. 고객 정보 출력\n");
        printf("4. 종료\n");
        printf("선택: ");
        scanf("%d", &choice);

        switch (choice) {
            case 1:
                if (num_customers < MAX_CUSTOMERS) {
                    add_customer(customers, &num_customers);
                } else {
                    printf("고객 최대 수에 도달했습니다.\n");
                }
                break;
            case 2:
                if (num_customers > 0) {
                    int index;
                    printf("포트폴리오를 추가할 고객 번호 (1~%d): ", num_customers);
                    scanf("%d", &index);
                    if (index >= 1 && index <= num_customers) {
                        add_portfolio(&customers[index - 1]);
                    } else {
                        printf("잘못된 입력입니다.\n");
                    }
                } else {
                    printf("등록된 고객이 없습니다.\n");
                }
                break;
            case 3:
                display_customers(customers, num_customers);
                break;
            case 4:
                free_memory(customers, num_customers);
                printf("프로그램 종료.\n");
                return 0;
            default:
                printf("올바른 번호를 입력하세요.\n");
        }
    }
}

// 고객 추가 함수
void add_customer(Customer* customers, int* num_customers) {
    printf("\n고객 %d 정보 입력\n", *num_customers + 1);
    printf("이름: ");
    scanf("%s", customers[*num_customers].name);
    printf("계좌 번호: ");
    scanf("%d", &customers[*num_customers].account_number);
    customers[*num_customers].stock_count = 0;
    customers[*num_customers].portfolio = NULL; // 초기 포트폴리오는 없음
    (*num_customers)++;
}

// 고객 포트폴리오 추가 및 수정 함수
void add_portfolio(Customer* customer) {
    int new_stocks;
    printf("\n%s님의 추가할 주식 개수 입력 (현재 %d개 보유, 최대 %d개 가능): ",
           customer->name, customer->stock_count, MAX_STOCKS);
    scanf("%d", &new_stocks);

    if (new_stocks <= 0 || customer->stock_count + new_stocks > MAX_STOCKS) {
        printf("잘못된 입력입니다.\n");
        return;
    }

    // 기존 포트폴리오에 추가할 크기만큼 메모리 재할당
    customer->portfolio = (Stock*) realloc(customer->portfolio,
                        (customer->stock_count + new_stocks) * sizeof(Stock));
    if (customer->portfolio == NULL) {
        printf("메모리 할당 실패.\n");
        return;
    }

    // 새로운 주식 정보 입력
    for (int i = 0; i < new_stocks; i++) {
        int idx = customer->stock_count + i;
        printf("\n주식 %d 정보 입력\n", idx + 1);
        printf("주식 이름: ");
        scanf("%s", customer->portfolio[idx].stock_name);
        printf("구매 가격: ");
        scanf("%lf", &customer->portfolio[idx].purchase_price);
        printf("현재 가격: ");
        scanf("%lf", &customer->portfolio[idx].current_price);
        printf("보유 수량: ");
        scanf("%d", &customer->portfolio[idx].quantity);
    }
    customer->stock_count += new_stocks;
}

// 고객 정보 출력 함수
void display_customers(Customer* customers, int num_customers) {
    printf("\n=== 고객 및 포트폴리오 정보 ===\n");
    for (int i = 0; i < num_customers; i++) {
        printf("\n고객 %d: %s (계좌번호: %d)\n", i + 1, customers[i].name, customers[i].account_number);
        printf("보유 주식 개수: %d\n", customers[i].stock_count);

        double total_purchase = 0, total_current = 0;
        for (int j = 0; j < customers[i].stock_count; j++) {
            Stock stock = customers[i].portfolio[j];
            double purchase_value = stock.purchase_price * stock.quantity;
            double current_value = stock.current_price * stock.quantity;
            total_purchase += purchase_value;
            total_current += current_value;

            printf("  [%d] %s - 구매가: %.2f, 현재가: %.2f, 수량: %d\n",
                   j + 1, stock.stock_name, stock.purchase_price, stock.current_price, stock.quantity);
        }
        printf("총 구매 금액: %.2f, 현재 가치: %.2f\n", total_purchase, total_current);
    }
}

// 동적 할당된 메모리 해제 함수
void free_memory(Customer* customers, int num_customers) {
    for (int i = 0; i < num_customers; i++) {
        free(customers[i].portfolio); // 고객 포트폴리오 해제
    }
}
