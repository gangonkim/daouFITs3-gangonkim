#define _CRT_SECURE_NO_WARNINGS

#include <stdio.h>
#include <string.h>

#define FILE_NAME "accounts.txt"

typedef struct {
    int id;
    char name[20];
    double balance;
} Account;

void addAccount() {
    FILE* file = fopen(FILE_NAME, "a");
    if (!file) {
        printf("파일을 열 수 없습니다.\n");
        return;
    }

    Account account;

    printf("계좌 ID: ");
    scanf("%d", &account.id);
    printf("고객 이름: ");
    scanf("%19s", account.name);
    printf("초기 잔액: ");
    scanf("%lf", &account.balance);

    fprintf(file, "ID: %d, Name: %s, Balance: %.2lf\n", account.id, account.name, account.balance);
    fclose(file);

    printf(">> 계좌 정보가 저장되었습니다.\n");
}

void printAccounts() {
    FILE* file = fopen(FILE_NAME, "r");
    if (!file) {
        printf("파일을 열 수 없습니다.\n");
        return;
    }

    char line[100];
    printf("\n[계좌 목록]\n");
    while (fgets(line, sizeof(line), file)) {
        printf("%s", line);
    }
    fclose(file);
}

void searchAccount() {
    FILE* file = fopen(FILE_NAME, "r");
    if (!file) {
        printf("파일을 열 수 없습니다.\n");
        return;
    }

    int searchId;
    printf("검색할 계좌 ID: ");
    scanf("%d", &searchId);

    char line[100];
    int found = 0;
    while (fgets(line, sizeof(line), file)) {
        int id;
        char name[20];
        double balance;
        if (sscanf(line, "ID: %d, Name: %19[^,], Balance: %lf", &id, name, &balance) == 3) {
            if (id == searchId) {
                printf("%s", line);
                found = 1;
                break;
            }
        }
    }
    if (!found) {
        printf("해당 ID의 고객을 찾을 수 없습니다.\n");
    }
    fclose(file);
}

void deposit() {
    int id;
    double amount;
    printf("입금할 계좌 ID: ");
    scanf("%d", &id);
    printf("입금할 금액: ");
    scanf("%lf", &amount);

    FILE* file = fopen(FILE_NAME, "r");
    FILE* temp = fopen("temp.txt", "w");
    if (!file || !temp) {
        printf("파일을 열 수 없습니다.\n");
        return;
    }

    char line[100];
    int found = 0;
    while (fgets(line, sizeof(line), file)) {
        int acc_id;
        char name[20];
        double balance;
        if (sscanf(line, "ID: %d, Name: %19[^,], Balance: %lf", &acc_id, name, &balance) == 3) {
            if (acc_id == id) {
                balance += amount;
                found = 1;
            }
        }
        fprintf(temp, "ID: %d, Name: %s, Balance: %.2lf\n", acc_id, name, balance);
    }
    fclose(file);
    fclose(temp);
    remove(FILE_NAME);
    rename("temp.txt", FILE_NAME);

    if (found) {
        printf("입금이 완료되었습니다.\n");
    } else {
        printf("계좌를 찾을 수 없습니다.\n");
    }
}

void withdraw() {
    int id;
    double amount;
    printf("출금할 계좌 ID: ");
    scanf("%d", &id);
    printf("출금할 금액: ");
    scanf("%lf", &amount);

    FILE* file = fopen(FILE_NAME, "r");
    FILE* temp = fopen("temp.txt", "w");
    if (!file || !temp) {
        printf("파일을 열 수 없습니다.\n");
        return;
    }

    char line[100];
    int found = 0;
    while (fgets(line, sizeof(line), file)) {
        int acc_id;
        char name[20];
        double balance;
        if (sscanf(line, "ID: %d, Name: %19[^,], Balance: %lf", &acc_id, name, &balance) == 3) {
            if (acc_id == id) {
                if (balance >= amount) {
                    balance -= amount;
                    found = 1;
                } else {
                    printf("잔액이 부족합니다.\n");
                }
            }
        }
        fprintf(temp, "ID: %d, Name: %s, Balance: %.2lf\n", acc_id, name, balance);
    }
    fclose(file);
    fclose(temp);
    remove(FILE_NAME);
    rename("temp.txt", FILE_NAME);

    if (found) {
        printf("출금이 완료되었습니다.\n");
    }
}

void transfer() {
    int from_id, to_id;
    double amount;
    printf("송금할 출처 계좌 ID: ");
    scanf("%d", &from_id);
    printf("송금할 대상 계좌 ID: ");
    scanf("%d", &to_id);
    printf("송금할 금액: ");
    scanf("%lf", &amount);

    FILE* file = fopen(FILE_NAME, "r");
    FILE* temp = fopen("temp.txt", "w");
    if (!file || !temp) {
        printf("파일을 열 수 없습니다.\n");
        return;
    }

    char line[100];
    int from_found = 0, to_found = 0;
    double from_balance = 0, to_balance = 0;
    while (fgets(line, sizeof(line), file)) {
        int acc_id;
        char name[20];
        double balance;
        if (sscanf(line, "ID: %d, Name: %19[^,], Balance: %lf", &acc_id, name, &balance) == 3) {
            if (acc_id == from_id) {
                if (balance >= amount) {
                    balance -= amount;
                    from_found = 1;
                    from_balance = balance;
                } else {
                    printf("잔액이 부족합니다.\n");
                }
            }
            if (acc_id == to_id) {
                balance += amount;
                to_found = 1;
                to_balance = balance;
            }
        }
        fprintf(temp, "ID: %d, Name: %s, Balance: %.2lf\n", acc_id, name, balance);
    }
    fclose(file);
    fclose(temp);
    remove(FILE_NAME);
    rename("temp.txt", FILE_NAME);

    if (from_found && to_found) {
        printf("송금이 완료되었습니다.\n");
    } else {
        printf("계좌를 찾을 수 없습니다.\n");
    }
}



int main(void) {
    char choice;
    while (1) {
        printf("\n[은행 계좌 시스템]\n");
        printf("1. 신규 계좌 개설 (N)\n");
        printf("2. 계좌 목록 출력 (L)\n");
        printf("3. 계좌 검색 (S)\n");
        printf("4. 입금 (D)\n");
        printf("5. 출금 (W)\n");
        printf("6. 계좌이체 (T)\n");
        printf("7. 종료 (Q)\n");
        printf("선택: ");
        scanf(" %c", &choice);

        switch (choice) {
            case 'N': case 'n': addAccount(); break;
            case 'L': case 'l': printAccounts(); break;
            case 'S': case 's': searchAccount(); break;
            case 'D': case 'd': deposit(); break;
            case 'W': case 'w': withdraw(); break;
            case 'T': case 't': transfer(); break;
            case 'Q': case 'q': return 0;
            default: printf("올바른 선택이 아닙니다.\n");
        }
    }
}
