#define _CRT_SECURE_NO_WARNINGS
#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define FILE_NAME "customers.txt"

typedef struct {
    int id;
    char name[50];
    double balance;
} Customer;

void addCustomer();
void listCustomers();
void searchCustomer();

int main() {
    char choice;
    while (1) {
        printf("\n[고객 정보 시스템]\n");
        printf("1. 신규 고객 추가 (N)\n");
        printf("2. 고객 목록 출력 (L)\n");
        printf("3. 고객 정보 검색 (S)\n");
        printf("4. 종료 (Q)\n");
        printf("선택: ");
        scanf(" %c", &choice);
        
        switch (choice) {
            case 'N': case 'n': addCustomer(); break;
            case 'L': case 'l': listCustomers(); break;
            case 'S': case 's': searchCustomer(); break;
            case 'Q': case 'q': return 0;
            default: printf("올바른 선택이 아닙니다.\n");
        }
    }
}

void addCustomer() {
    FILE* file = fopen(FILE_NAME, "a");
    if (!file) {
        printf("파일을 열 수 없습니다.\n");
        return;
    }
    
    Customer c;
    printf("고객 ID: ");
    scanf("%d", &c.id);
    printf("고객 이름: ");
    scanf("%s", c.name);
    printf("잔액: ");
    scanf("%lf", &c.balance);
    
    fprintf(file, "ID: %d, Name: %s, Balance: %.2lf\n", c.id, c.name, c.balance);
    fclose(file);
    
    printf(">> 고객 정보가 저장되었습니다.\n");
}

void listCustomers() {
    FILE* file = fopen(FILE_NAME, "r");
    if (!file) {
        printf("파일을 열 수 없습니다.\n");
        return;
    }
    
    char line[100];
    printf("\n[고객 목록]\n");
    while (fgets(line, sizeof(line), file)) {
        printf("%s", line);
    }
    fclose(file);
}

void searchCustomer() {
    FILE* file = fopen(FILE_NAME, "r");
    if (!file) {
        printf("파일을 열 수 없습니다.\n");
        return;
    }
    
    int searchId;
    printf("검색할 고객 ID: ");
    scanf("%d", &searchId);
    
    char line[100];
    int found = 0;
    while (fgets(line, sizeof(line), file)) {
        if (strstr(line, "ID:")) {
            int id;
            sscanf(line, "ID: %d,", &id);
            if (id == searchId) {
                printf("%s", line);
                found = 1;
                break;
            }
        }
    }
    
    if (!found) {
        printf("해당 ID의 고객을 찾을 수 없습니다.\n");
    }
    fclose(file);
}

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

typedef enum {
    CASH, CARD, CRYPTO
} TransactionType;

typedef union {
    struct {
        int amount;
    } cash;

    struct {
        char number[20];
        int halbumonth;
        int amount;
    } card;

    struct {
        char address[20];
        int amount;
    } crypto;

} TransactionData;


typedef struct {
    char name[20];
    TransactionType type;
    TransactionData data;
} Transaction;

Transaction* transactions = NULL;
int transactionCount = 0;

void addTransaction() {
    transactions = (Transaction*) realloc(transactions, (transactionCount + 1) * sizeof(Transaction));
    Transaction* t = &transactions[transactionCount];
    printf("고객 이름: \n");
    scanf_s("%s", t->name, 20);
    printf("거래 유형 (0: 현금, 1: 카드, 2: 가상화폐): ");
    scanf_s("%d", (int*)&t->type);


    switch (t->type) {
        case CASH:
            printf("거래 금액: ");
            scanf_s("%d", &t->data.cash.amount);
        break;
        case CARD:
            printf("카드 번호: ");
            scanf_s("%s\n", t->data.card.number);
            printf("할부 개월 수: ");
            scanf_s("%d\n", &t->data.card.halbumonth);
            printf("거래 금액: ");
            scanf_s("%d\n", &t->data.card.amount);
        break;
        case CRYPTO:
            printf("지갑 주소: ");
            scanf_s("%s", t->data.crypto.address);
            printf("거래 금액: ");
            scanf_s("%lf", &t->data.crypto.amount);
        break;
        default:
            printf("잘못된 입력입니다.\n");
        return;
    }

    transactionCount++;
    printf("거래가 추가되었습니다.\n");
};

void printTransactions() {
    for (int i = 0; i < transactionCount; i++) {
        switch (transactions[i].type) {
            case CASH:
                printf("고객 이름: %s | ", transactions[i].name);
                printf("거래 유형: 현금 거래");
                printf("금액 :%d\n", transactions[i].data.cash.amount);
            break;
            case CARD:
                printf("고객 이름: %s | ", transactions[i].name);
                printf("거래 유형: 카드 거래 | ");
                printf("카드번호 :%s | ", transactions[i].data.card.number);
                printf("할부 개월 :%s | ", transactions[i].data.card.halbumonth);
                printf("금액 :%d\n", transactions[i].data.card.amount);
            break;;
            case CRYPTO:
                printf("고객 이름: %s | ", transactions[i].name);
                printf("거래 유형: 가상화폐 거래  | ");
                printf("지갑 주소: %s | ", transactions[i].data.crypto.address);
                printf("금액: %d\n", transactions[i].data.crypto.amount);
            break;
        }
    }
}

int main(void) {

    while (1) {
        int choice;
        printf("1. 거래 추가\n");
        printf("2. 거래 내역 출력\n");
        printf("3. 종료\n");
        scanf("%d", &choice);
        switch (choice) {
            case 1:
                addTransaction();
            break;
            case 2:
                printTransactions();
                    break;
            case 3:
                free(transactions);
            return 0;
        }
    }

}
