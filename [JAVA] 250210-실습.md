package first;


import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;
import java.sql.ResultSet;

//책 제목에 대한 키워드를 입력받아서 이 키워드를 포함하고 있는 책 제목을 검색, 해당 책의 제모그 저자, 가격 정보 출력
public class SecondConnect {
    public static void main(String[] args) {
        //JDBC는 크게 6개의 단계로 사용

        // 1. (JDBC) DRIVER로딩 - 프로그램에서 딱 한번만 수행
        try {
            Class.forName("oracle.jdbc.driver.OracleDriver");

            // 2. 데이터베이스 연결 - 연결 성공의 증표로 객체가 생성됨
            // 객체의 Class는 java.sql.Connection
            Connection con;

            // 연결을 위해서는 3가지 정보가 필요
            // 1. 어떤 데이터베이스에 접속하는지에 대한 정보(문자열로 표현) => JDBC URL
            //  jdbc:데이터베이스 종류:접속방식(thin or oci):컴터주소:포트번호:xe
            String url = "jdbc:oracle:thin:@localhost:1521:xe";
            // 2.id
            String id = "C##DEV";
            // 3.password
            String pw = "sys1234";


            con = DriverManager.getConnection(url, id, pw);

            String keyword = "여행";

            //SQL 작성 -> Statement 생성
            //Statement의 종류:
            // 1. 일반 Statement -> 단순하지만 기능상으로 부족
            // 2. PreparedStatement -> 많이 사용, 코드 작성 편리하고 효율이 좋음, 보안(SQL Injection 방지)
            // 3. CallableStatement -> Stored 프로시저를 호출하기 위해 사용
            // 4. 쿼리 실행
            //      excuteUpdate(): SQL구문이 insert, update, delete일 경우 사용
            //      excuteQuery(): SQL구문이 select일 경우 사용
            String sql = "select btitle, bauthor, bprice from book where btitle like ?";
            PreparedStatement pstmt = con.prepareStatement(sql);
            pstmt.setString(1, '%' + keyword + '%');

            // 5. 결과 확인
            //      excuteUpdate(): 영향을 받은 행 수 리턴
            //      excuteQuery(): ResultSet이 리턴
            //      rs.next -> 다음 행 가져오는 메서드, 성공하면 true 리턴
            ResultSet rs = pstmt.executeQuery();
            while (rs.next()) {
                System.out.println(rs.getString("btitle")+" "
                        +rs.getString("bauthor")+" "
                        +rs.getString("bprice"));
            }

            // 6. 리소스 해제
            // Connection, PrepareStatement, ResultSet에서 사용된 자원을 생성 순서의 반대로 해제
            rs.close();
            pstmt.close();
            con.close();

        } catch(Exception e) {
            System.out.println(e.getMessage());
        }
    }
}

package first;

import java.sql.Connection;
import java.sql.DriverManager;
import java.sql.PreparedStatement;

public class TransactionTest {
    public static void main(String[] args) {
        Connection con = null;

        try {
            Class.forName("oracle.jdbc.driver.OracleDriver");

            String url = "jdbc:oracle:thin:@localhost:1521:xe";
            String id = "C##DEV";
            String pw = "sys1234";
            con = DriverManager.getConnection(url, id, pw);
            //기본적으로 auto commit 모드로 Connection이 실행

            con.setAutoCommit(false); // transaction 설정
            //명시적으로 커밋이나 롤백이 호출되면 트랜잭션이 종료
            //정상적으로 트랜잭션이 종료되면 커밋으로 간주

            String sql = "insert into memberTbl values(?,?)";
            PreparedStatement pstmt = con.prepareStatement(sql);
            pstmt.setString(1, "1");
            pstmt.setString(2, "홍길동");
            if(pstmt.executeUpdate() == 1) {
                System.out.println("업데이트 성공");
            } else {
                System.out.println("업데이트 실패");
            }
            //con.rollback(); // 지금까지 수행한 SQL작업 무효화
            con.commit(); // 커밋
            pstmt.close();
            con.close();




        } catch (Exception e) {
            e.printStackTrace();
        }
    }
}
