#include <stdio.h>
#include <stdlib.h>

#define MAX_ACCOUNTS 10  // 최대 계좌 개수

typedef struct {
    int accountNumber;  // 계좌 번호
    double balance;     // 계좌 잔액
} Account;

typedef struct {
    char name[30];   // 고객 이름
    int age;         // 나이
    int customerID;  // 고객 ID
    int accountCount;  // 현재 보유 계좌 수
    Account accounts[MAX_ACCOUNTS]; // 고객 계좌 목록
} Customer;

// 함수 선언
void addAccount(Customer *customer);
void deleteAccount(Customer *customer);
void viewCustomerInfo(const Customer *customer);
void displayMenu();

int main() {
    Customer customer = {"홍길동", 30, 1001, 0};  // 고객 정보 초기화
    int choice;

    while (1) {
        displayMenu();
        printf("원하는 작업을 선택하세요: ");
        scanf("%d", &choice);

        switch (choice) {
            case 1:
                addAccount(&customer);
                break;
            case 2:
                deleteAccount(&customer);
                break;
            case 3:
                viewCustomerInfo(&customer);
                break;
            case 4:
                printf("프로그램을 종료합니다.\n");
                return 0;
            default:
                printf("잘못된 입력입니다. 다시 선택하세요.\n");
        }
    }
    return 0;
}

// 메뉴 출력 함수
void displayMenu() {
    printf("\n1. 계좌 추가\n");
    printf("2. 계좌 삭제\n");
    printf("3. 고객 정보 보기\n");
    printf("4. 종료\n");
}

// 계좌 추가 함수
void addAccount(Customer *customer) {
    if (customer->accountCount >= MAX_ACCOUNTS) {
        printf("더 이상 계좌를 추가할 수 없습니다.\n");
        return;
    }

    int accountNumber;
    double balance;

    printf("추가할 계좌 번호를 입력하세요: ");
    scanf("%d", &accountNumber);
    printf("계좌의 초기 잔액을 입력하세요: ");
    scanf("%lf", &balance);

    customer->accounts[customer->accountCount].accountNumber = accountNumber;
    customer->accounts[customer->accountCount].balance = balance;
    customer->accountCount++;

    printf("계좌가 성공적으로 추가되었습니다.\n");
}

// 계좌 삭제 함수
void deleteAccount(Customer *customer) {
    if (customer->accountCount == 0) {
        printf("삭제할 계좌가 없습니다.\n");
        return;
    }

    int accountNumber;
    printf("삭제할 계좌 번호를 입력하세요: ");
    scanf("%d", &accountNumber);

    int index = -1;
    for (int i = 0; i < customer->accountCount; i++) {
        if (customer->accounts[i].accountNumber == accountNumber) {
            index = i;
            break;
        }
    }

    if (index == -1) {
        printf("해당 계좌 번호를 찾을 수 없습니다.\n");
        return;
    }

    for (int i = index; i < customer->accountCount - 1; i++) {
        customer->accounts[i] = customer->accounts[i + 1];
    }
    customer->accountCount--;

    printf("계좌가 성공적으로 삭제되었습니다.\n");
}

// 고객 정보 출력 함수
void viewCustomerInfo(const Customer *customer) {
    printf("\n고객: %s\n", customer->name);
    printf("나이: %d\n", customer->age);
    printf("고객 ID: %d\n", customer->customerID);
    printf("계좌 수: %d\n", customer->accountCount);

    for (int i = 0; i < customer->accountCount; i++) {
        printf("계좌 %d: 잔액 %.2f\n", customer->accounts[i].accountNumber, customer->accounts[i].balance);
    }
}

#include <stdio.h>
#include <stdlib.h>
#include <string.h>

#define MAX_INVESTMENTS 10  // 최대 투자 상품 개수

// 투자 상품 구조체
typedef struct {
    char name[30];           // 종목 이름
    double investment_amount; // 투자 금액
    double return_rate;       // 수익률 (예: 0.1 -> 10%)
} Investment;

// 고객 구조체
typedef struct {
    char name[30];                  // 고객 이름
    int num_investments;             // 투자 상품 개수
    Investment *investments[MAX_INVESTMENTS]; // 투자 상품 배열 (포인터 사용)
} Customer;

// 함수 선언
void add_investment(Customer *customer, const char *name, double amount, double rate);
void calculate_profit(const Customer *customer);

int main() {
    Customer customer;

    // 고객 정보 입력
    printf("고객 이름은? ");
    scanf("%s", customer.name);
    customer.num_investments = 0;

    int num;
    printf("투자 상품 개수는? ");
    scanf("%d", &num);

    if (num > MAX_INVESTMENTS) {
        printf("최대 %d개의 투자 상품만 등록할 수 있습니다.\n", MAX_INVESTMENTS);
        return 1;
    }

    // 투자 상품 입력
    for (int i = 0; i < num; i++) {
        char name[30];
        double amount, rate;

        printf("투자 상품 %d을 입력하세요 (회사명, 투자 금액, 수익률):\n", i + 1);
        scanf("%s %lf %lf", name, &amount, &rate);

        add_investment(&customer, name, amount, rate / 100); // 입력값을 0.XX 형태로 변환
    }

    // 투자 상품별 수익 출력
    calculate_profit(&customer);

    // 동적 할당된 메모리 해제
    for (int i = 0; i < customer.num_investments; i++) {
        free(customer.investments[i]);
    }

    return 0;
}

// 투자 상품 추가 함수
void add_investment(Customer *customer, const char *name, double amount, double rate) {
    if (customer->num_investments >= MAX_INVESTMENTS) {
        printf("더 이상 투자 상품을 추가할 수 없습니다.\n");
        return;
    }

    Investment *newInvestment = (Investment *)malloc(sizeof(Investment));
    if (newInvestment == NULL) {
        printf("메모리 할당 실패\n");
        return;
    }

    strcpy(newInvestment->name, name);
    newInvestment->investment_amount = amount;
    newInvestment->return_rate = rate;

    customer->investments[customer->num_investments] = newInvestment;
    customer->num_investments++;

    printf("%s 투자 상품이 추가되었습니다.\n", name);
}

// 수익 계산 및 출력 함수
void calculate_profit(const Customer *customer) {
    printf("\n각 투자 상품의 수익:\n");

    for (int i = 0; i < customer->num_investments; i++) {
        Investment *inv = customer->investments[i];
        double profit = inv->investment_amount * inv->return_rate;
        printf("%s 투자 상품의 수익: %.2f\n", inv->name, profit);
    }
}

#include <stdio.h>
#include <time.h>
#include <string.h>

#define MAX_DEPOSITS 5

// 정기 예금 구조체 정의
typedef struct {
    char name[30];            // 고객 이름
    char account_number[11];  // 계좌 번호 (문자열, 10자리)
    struct tm start_date;     // 예금 시작 날짜
    int deposit_period;       // 예치 기간 (개월 단위)
    double deposit_amount;    // 예치 금액
} FixedDeposit;

// 예금 계좌 배열 (정적 할당)
FixedDeposit deposits[MAX_DEPOSITS] = {
    {"김철수", "1234567890", {0, 0, 0, 10, 0, 124}, 12, 5000000},
    {"이영희", "9876543210", {0, 0, 0, 1, 6, 123}, 24, 3000000},
    {"박민수", "5678901234", {0, 0, 0, 15, 2, 123}, 18, 2000000},
    {"조수빈", "3456789012", {0, 0, 0, 5, 8, 123}, 6, 1000000},
    {"정하나", "6789012345", {0, 0, 0, 20, 10, 123}, 12, 7000000}
};

//정기 예금 만기일 계산 함수
time_t calculate_maturity_date(struct tm start_date, int period) {
    start_date.tm_mon += period; // 예치 기간을 추가 (개월 단위)
    mktime(&start_date); // 날짜 업데이트
    return mktime(&start_date);
}

//만기일이 빠른 순으로 정렬하는 함수
void sort_by_maturity_date(FixedDeposit arr[], int size) {
    for (int i = 0; i < size - 1; i++) {
        for (int j = 0; j < size - i - 1; j++) {
            time_t date1 = calculate_maturity_date(arr[j].start_date, arr[j].deposit_period);
            time_t date2 = calculate_maturity_date(arr[j + 1].start_date, arr[j + 1].deposit_period);
            if (date1 > date2) {
                FixedDeposit temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
            }
        }
    }
}

//현재 날짜와 비교하여 만기 여부 확인
void check_maturity(FixedDeposit arr[], int size) {
    time_t now = time(NULL); // 현재 날짜 가져오기

    printf("\n 만기 여부 확인 결과:\n");
    for (int i = 0; i < size; i++) {
        time_t maturity_date = calculate_maturity_date(arr[i].start_date, arr[i].deposit_period);
        double days_left = difftime(maturity_date, now) / (60 * 60 * 24);

        printf("고객: %s (계좌번호: %s)\n", arr[i].name, arr[i].account_number);
        if (days_left <= 0) {
            printf("만기 도래! 출금 가능\n");
        } else {
            printf("만기까지 남은 일수: %.0f일\n", days_left);
        }
        printf("---------------------------\n");
    }
}

// 정렬된 예금 리스트 출력
void display_deposits(FixedDeposit arr[], int size) {
    printf("\n 정렬된 예금 리스트 (만기일 빠른 순):\n");
    for (int i = 0; i < size; i++) {
        struct tm maturity_date = arr[i].start_date;
        maturity_date.tm_mon += arr[i].deposit_period; // 만기일 계산
        mktime(&maturity_date); // 날짜 업데이트

        printf("고객: %s, 계좌번호: %s\n", arr[i].name, arr[i].account_number);
        printf("예치 금액: %.2f 원\n", arr[i].deposit_amount);
        printf("예금 시작일: %d-%02d-%02d\n", arr[i].start_date.tm_year + 1900, arr[i].start_date.tm_mon + 1, arr[i].start_date.tm_mday);
        printf("만기 예정일: %d-%02d-%02d\n", maturity_date.tm_year + 1900, maturity_date.tm_mon + 1, maturity_date.tm_mday);
        printf("---------------------------\n");
    }
}

int main() {
    // 정기 예금 만기일 기준 정렬
    sort_by_maturity_date(deposits, MAX_DEPOSITS);

    // 정렬된 예금 리스트 출력
    display_deposits(deposits, MAX_DEPOSITS);

    // 만기 여부 확인
    check_maturity(deposits, MAX_DEPOSITS);

    return 0;
}

#define STOCKS 3
typedef struct {

    char name[20];

} Stocks;

   66300, 65900, 65550, 65800, 66050, 66350, 66700, 66800, 67000, 66900, 67200, 67300,
   67100, 67050, 67250, 67400, 67600, 67500, 67450, 67650, 67800, 67900}},
    {"테슬라", "TSLA", 50, {210, 215, 220, 218, 222, 225, 230, 228, 232, 235, 238, 240,
   245, 250, 248, 246, 244, 241, 238, 235, 233, 231, 228, 226, 224, 223, 220, 218, 215, 212}},
    {"애플", "AAPL", 80, {180, 182, 184, 183, 185, 186, 188, 187, 189, 191, 193, 195, 198,
   200, 202, 203, 205, 207, 209, 210, 212, 213, 214, 215, 217, 218, 219, 220, 221, 222}}
};

void calculate_average() {

    for (int i = 0; i < STOCKS; i++) {
        int sum = 0;
        for (int j = 0; j < 30; j++) {
            sum += stocks[i].dayprice[j];
        }
        stocks[i].avgprice = sum / 30;
    };
};

void calculate_volatility() {
    for (int i = 0; i < STOCKS; i++) {
        double sum = 0.0;
        for (int j = 0; j < 30; j++) {
            sum += pow(stocks[i].dayprice[j] - stocks[i].avgprice, 2);
        }
        stocks[i].volatility = sqrt(sum / 30.0);
    };
};

char* categorize_risk(int volatility) {
    if (volatility <= 5) {
        return "낮음";
    } else if (volatility <= 10) {
        return "보통";
    } else {
        return"높음";
    }
}

void display_portfolio() {
    printf("[고객 주식 포트폴리오 위험 분석]\n");
    for (int i = 0; i < STOCKS; i++) {
        printf("종목이름: %s | 보유주식: %d | 변동성: %lf | 위험수준: %s\n", stocks[i].name, stocks[i].count, stocks[i].volatility, categorize_risk(stocks[i].volatility));
    }
}


int main(void) {
    calculate_average();
    calculate_volatility();

    display_portfolio();
    return 0;
}
